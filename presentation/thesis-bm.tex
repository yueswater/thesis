\section{Introduction}

\begin{frame}{Research Question}
  \begin{itemize}
    \item \citet{fu2006}: in a Tullock contest with private information,
      the informed player prefers to move \emph{second} $\Rightarrow$
      revealing type early is costly.
    \item But this rests on $\rho = 1$: prize values are
      \emph{perfectly correlated}.
    \item \textbf{Gap}: does the late-mover preference survive at $\rho < 1$,
      or is it an artifact of full correlation?
    \item \textbf{This paper}: introduce $\rho \in [0,1]$ $\Rightarrow$
      at $\rho = 0$ the signal channel is closed;
      at $\rho = 1$ Fu's result is recovered as the limiting case.
  \end{itemize}

  \vfill
  \begin{center}
    \begin{tikzpicture}[scale=0.85, font=\footnotesize]
      \draw[->, thick] (0,0) -- (7.2,0) node[right] {$\rho$};
      \foreach \x/\lab in {0/{$0$}, 2.4/{$\rhodag$}, 4.8/{$\rhosep$}, 6.8/{$1$}}{
        \draw[thick] (\x,0.08) -- (\x,-0.08) node[below] {\lab};
      }
      \node[below=0.45cm, font=\scriptsize] at (6.8,0) {\citet{fu2006}};
    \end{tikzpicture}
  \end{center}
\end{frame}

\begin{frame}{What This Paper Does}
  \begin{itemize}
    \setlength{\itemsep}{1.2em}
    \item \textbf{\textit{i.i.d.} benchmark} ($\rho = 0$) $\Rightarrow$
      isolates timing from information spillover.
    \item \textbf{Partial correlation} ($0 < \rho < 1$) as the main model\\[5pt]
      $\Rightarrow$ $\rho = 1$ is the endpoint, not a separate model.
    \item Characterizes undistorted vs.\ Riley distorted separation,
      and their transition as $\rho$ rises.
    \item Links subgame equilibria to endogenous timing:
      a structural asymmetry between $I$ and $U$.
  \end{itemize}
\end{frame}

\section{Motivation and Benchmark}

\begin{frame}{Model Setup}
  \begin{itemize}
    \item Two players: informed ($I$) and uninformed ($U$)
    \item Types: $V_I, V_U \in \{v_H, v_L\}$,
      $\Pr(V_k = v_H) = q \in (0,1)$
    \item \textbf{Benchmark assumption}: $V_I \perp V_U$
      (\textit{i.i.d.}, $\rho = 0$)
    \item Define $\bar v \equiv q v_H + (1-q) v_L = \E[V_k]$
  \end{itemize}

  \begin{center}
    \small
    \begin{tabular}{@{}lcc@{}}
      \toprule
      & $V_U = v_H$ & $V_U = v_L$ \\
      \midrule
      $V_I = v_H$ & $q^2$ & $q(1-q)$ \\
      $V_I = v_L$ & $q(1-q)$ & $(1-q)^2$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \begin{itemize}
    \item Three-stage timing: Stage 0 (choose timing) $\Rightarrow$
      Nature draws types $\Rightarrow$ effort subgame
    \item Three regimes: SS (simultaneous), UI ($U$ moves first),
      IU ($I$ moves first)
  \end{itemize}
\end{frame}

\begin{frame}{Equilibrium Payoffs across Regimes}
  Composite coefficients:
  $\hat A \equiv \bar v\,\E\bigl[V_I^{-1/2}\bigr]$,
  $\hat B \equiv \bar v\,\E\bigl[V_I^{-1}\bigr] \ge 1$ (Jensen).

  \begin{center}
    \small
    \renewcommand{\arraystretch}{1.8}
    \begin{tabular}{@{}lccc@{}}
      \toprule
      Regime & $x_U^*$ & $\widetilde\pi_I$ & $\widetilde\pi_U$\\
      \midrule
      SS
        & $\dfrac{\hat A^2}{(1+\hat B)^2}$
        & $\E\Bigl[\bigl(\sqrt{V_I} - \tfrac{\hat A}{1+\hat B}\bigr)^2\Bigr]$
        & $\dfrac{\hat A^2 \hat B}{(1+\hat B)^2}$ \\
      UI
        & $\dfrac{\hat A^2}{4}$
        & $\E\Bigl[\bigl(\sqrt{V_I} - \tfrac{\hat A}{2}\bigr)^2\Bigr]$
        & $\dfrac{\hat A^2}{4}$ \\
      IU (sep.)
        & $\dfrac{v(2\bar v - v)}{4\bar v}$
        & $\dfrac{\E[V_I^2]}{4\bar v}$
        & $\dfrac{\E\bigl[(2\bar v - V_I)^2\bigr]}{4\bar v}$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  $\Rightarrow$ \textbf{First-mover advantage for $U$}:
  $\widetilde\pi_U^{UI} - \widetilde\pi_U^{SS}
    = \dfrac{\hat A^2 (1-\hat B)^2}{4(1+\hat B)^2} \ge 0$.
\end{frame}

\begin{frame}{Stage-0 Timing Game}
  $\widetilde\pi_U^{UI} \ge \widetilde\pi_U^{SS}$ always, yet
  $\widetilde\pi_I^{SS} > \widetilde\pi_I^{UI}$ whenever $\hat B > 1$ $\Rightarrow$
  $U$ wants to lead, but $I$ refuses to follow: UI is never a
  pure-strategy equilibrium. The real choice is between IU and SS.
  Define the informed player's unilateral deviation gaps:
  \[
    D_I \equiv \widetilde\pi_I^{IU} - \widetilde\pi_I^{SS},
    \qquad
    D_U \equiv \widetilde\pi_U^{IU} - \widetilde\pi_U^{SS}.
  \]

  \begin{block}{Proposition (Stage-0 Pure-strategy Equilibria, Benchmark)}
    \begin{itemize}\setlength{\itemsep}{0.2em}
      \item UI is never a pure-strategy equilibrium
      \item IU \; iff \; $D_I \ge 0$
      \item SS \; iff \; $D_U \le 0$
      \item if $D_I < 0 < D_U$: \emph{no} pure-strategy equilibrium exists
    \end{itemize}
  \end{block}

  $\Rightarrow$ Since $\rho = 0$ closes the signal channel, this split is driven
  purely by who benefits from committing first $\Rightarrow$ no signaling distortion.
\end{frame}

\section{Main Model}

\begin{frame}{Correlation \& Posterior Expected Valuation}
  $\rho \equiv \operatorname{corr}(V_I, V_U) \in [0,1]$: correlation between
  $I$'s and $U$'s prize values, with marginals fixed at $(q, 1-q)$.

  \begin{block}{Lemma (Posterior Expected Valuation)}
    Given posterior belief $\mu = \Pr(V_I = v_H \mid x_I)$,
    \[
      \VU(\mu) = v_L + \bigl[q + \rho(\mu - q)\bigr]\Delta V,
      \qquad
      \frac{d\VU(\mu)}{d\mu} = \rho\,\Delta V > 0.
    \]
  \end{block}

  \begin{center}
    \footnotesize
    \begin{tabular}{@{}llc@{}}
      \toprule
      Posterior $\mu$ & Interpretation & $\VU(\mu)$ \\
      \midrule
      $\mu = 1$ & $I$ is high type & $v_L + [q+\rho(1-q)]\Delta V$ \\
      $\mu = 0$ & $I$ is low type & $v_L + q(1-\rho)\Delta V$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  $\Rightarrow$ When $\rho > 0$, $I$'s type is a signal about $U$'s own prize
  $\Rightarrow$ the channel is live.
\end{frame}

\begin{frame}{SS and UI under Partial Correlation}
  Composite coefficients $\Rightarrow$ replace $\bar v$ with $\VU(1)$, $\VU(0)$:
  \[
    \tilde A \equiv q\,\frac{\VU(1)}{\sqrt{v_H}}
      + (1-q)\,\frac{\VU(0)}{\sqrt{v_L}},
    \qquad
    \tilde B \equiv q\,\frac{\VU(1)}{v_H}
      + (1-q)\,\frac{\VU(0)}{v_L}.
  \]

  \begin{block}{Proposition (First-mover Advantage Robust)}
    $\widetilde\pi_U^{UI} \ge \widetilde\pi_U^{SS}$ for all $\rho \in [0,1]$.
  \end{block}
\end{frame}

\begin{frame}{When Does Separation Hold?}
  Undistorted separating candidate in the IU subgame, and its IC condition:
  \[
    x_H = \frac{v_H^2}{4\VU(1)},
    \quad
    x_L = \frac{v_L^2}{4\VU(0)};
    \qquad
    \frac{\VU(1)}{\VU(0)} \le R
    \equiv \frac{v_H^2}{v_L(2v_H - v_L)}.
  \]

  \begin{block}{Proposition (Separation Threshold)}
    Undistorted separation holds iff $\rho \le \rhosep$, where
    \[
      \rhosep = \frac{(R-1)\,\VU(q)}{(1-q+Rq)\,\Delta V}.
    \]
  \end{block}

  \begin{itemize}
    \item At $\rho = 0$: strictly satisfied; at $\rho = 1$: violated
    \item $\VU(1)/\VU(0)$ is strictly increasing in $\rho$, so the IC
      condition reduces to a single threshold $\rhosep$
  \end{itemize}
\end{frame}

\begin{frame}{Riley Distortion}
  When $\rho > \rhosep$, define
  \[
    \alpha \equiv 1 - \frac{\VU(0)}{\VU(1)} \in (0,1),
    \qquad
    z_H^- \equiv \frac{v_H}{2\sqrt{\VU(0)}}\bigl(1 - \sqrt{\alpha}\bigr).
  \]

  \begin{itemize}
    \item High type keeps undistorted effort $x_H$;
      low type distorts \emph{down} to $x_L^{R} = (z_H^-)^2$
  \end{itemize}

  \begin{block}{Corollary (Region Structure of the IU Subgame)}
    \begin{itemize}\setlength{\itemsep}{0.2em}
      \item $\rho \le \rhosep$: undistorted separation $\Rightarrow$ no signaling cost
      \item $\rho > \rhosep$: Riley distortion $\Rightarrow$ cost borne by the low type alone
      \item $\rho \ge \rhopool$ may also admit \emph{pooling}:
        $\mu = q$, the signal is lost $\Rightarrow$
        coexists with whichever separating type applies at that $\rho$
    \end{itemize}
  \end{block}

  $\Rightarrow$ As $\rho$ rises, the low type's effort no longer supports
  separation, so it must distort downward.
\end{frame}

\begin{frame}{Stage-0 Payoff Matrix}
  \textbf{Key point}: at stage 0, types are unrealized $\Rightarrow$
  players compare \emph{ex ante} expected payoffs.

  \begin{columns}
    \begin{column}{.5\textwidth}
      \begin{center}
        \renewcommand{\arraystretch}{1.0}
        \begin{tabular}{@{}lcc@{}}
          \toprule
          & $U$: Early & $U$: Late \\
          \midrule
          $I$: Early
            & $\bigl(\widetilde\pi_I^{SS},\, \widetilde\pi_U^{SS}\bigr)$
            & $\bigl(\widetilde\pi_I^{IU},\, \widetilde\pi_U^{IU}\bigr)$ \\
          $I$: Late
            & $\bigl(\widetilde\pi_I^{UI},\, \widetilde\pi_U^{UI}\bigr)$
            & $\bigl(\widetilde\pi_I^{SS},\, \widetilde\pi_U^{SS}\bigr)$ \\
          \bottomrule
        \end{tabular}
      \end{center}
    \end{column}

    \begin{column}{.5\textwidth}
        \footnotesize
        \begin{align*}
          D_I^{UI} &\equiv \widetilde\pi_I^{UI} - \widetilde\pi_I^{SS}, \quad
          D_I^{IU} \equiv \widetilde\pi_I^{IU} - \widetilde\pi_I^{SS}, \\
          D_U^{UI} &\equiv \widetilde\pi_U^{UI} - \widetilde\pi_U^{SS}, \quad
          D_U^{IU} \equiv \widetilde\pi_U^{IU} - \widetilde\pi_U^{SS}.
        \end{align*}
    \end{column}
  \end{columns}

  \begin{block}{Lemma (Stage-0 Pure-strategy Equilibria)}
    \begin{itemize}\setlength{\itemsep}{0.1em}
      \item UI \; iff \; $D_I^{UI} \ge 0$ \emph{and} $D_U^{UI} \ge 0$
      \item SS \; iff \; $D_I^{UI} \le 0$ \emph{and} $D_U^{IU} \le 0$
      \item IU \; iff \; $D_I^{IU} \ge 0$ \emph{and} $D_U^{IU} \ge 0$
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Endogenous Timing under Correlated Type Structure}
  \small
  \begin{itemize}\setlength{\itemsep}{0.2em}
    \item $U$'s side (recap): $D_U^{UI} \ge 0$ for all
      $\rho \in [0,1]$ $\Rightarrow$ $U$ always wants to lead against a late-moving $I$.
    \item $I$'s side, new: $D_I^{UI} \le 0$ throughout
      the interior, with equality \emph{only} at $\rho = 1$ $\Rightarrow$
      $I$ (almost) never wants to lead via UI.
  \end{itemize}

  \begin{block}{Theorem (Stage-0 Timing Equilibrium)}
    \begin{itemize}\setlength{\itemsep}{0.1em}
      \item $\rho \le \rhodag$: IU is the unique equilibrium
        ($D_I^{IU} \ge 0$, $D_U^{IU} \ge 0$)
      \item $\rhodag < \rho < 1$: IU and UI are both ruled out;
        SS iff $D_U^{IU}(\rho) \le 0$;
        \emph{otherwise no pure-strategy equilibrium exists}
      \item $\rho = 1$: UI is the unique equilibrium
        $\Rightarrow$ \citet{fu2006}'s endpoint
    \end{itemize}
  \end{block}

  $\Rightarrow$ The gap beyond $\rhodag$ is not a clean flip to SS $\Rightarrow$
  a band with \emph{no} pure-strategy timing equilibrium can open up
  (numerical confirmation next).
\end{frame}

\section{Welfare Analysis}

\begin{frame}{Welfare Framework}
  Social welfare = allocative value minus effort dissipation:
  $\widetilde W = \widetilde G - \widetilde T$,
  where $\widetilde G = \E[p_I V_I + p_U V_U]$
  and $\widetilde T = \E[x_I + x_U]$.

  \begin{center}
    \footnotesize
    \renewcommand{\arraystretch}{1.5}
    \begin{tabular}{@{}lcl@{}}
      \toprule
      Equilibrium type & Expected dissipation $\widetilde T$
        & Sensitivity to $\rho$ \\
      \midrule
      SS
        & $\dfrac{\tilde A}{1+\tilde B}\,\E\bigl[\sqrt{V_I}\bigr]$
        & ambiguous ($\tilde A$, $\tilde B$ both fall) \\
      UI
        & $\dfrac{\tilde A}{2}\,\E\bigl[\sqrt{V_I}\bigr]$
        & strictly decreasing (via $\tilde A$) \\
      IU undistorted sep.
        & $\dfrac{\E[V_I]}{2}$
        & immune to $\rho$ \\
      IU Riley distorted
        & $\dfrac{v_H}{2}\bigl[1-(1-q)\sqrt{\alpha}\bigr]$
        & varies via $\alpha$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  {\small $\Rightarrow$ Pooling ($\rho\ge\rhopool$) is excluded from the welfare
  comparison: $D_I^{IU,\mathrm{pool}}\le 0$ means $I$ never prefers IU
  under pooling, so it cannot sustain endogenous timing and no separate
  pooling welfare formula is derived $\Rightarrow$ only IU undistorted
  separation's dissipation is insulated from $\rho$.}
\end{frame}

\begin{frame}{Three Effects of $\rho$ on Welfare}
  \begin{itemize}
    \item \textbf{Allocative effect}: $\rho\uparrow$ $\Rightarrow$
      $\VU(1)$ and $\VU(0)$ diverge $\Rightarrow$ win probabilities and prize
      allocation shift across paths; the welfare effect is not
      sign-definite a priori
    \item \textbf{Dissipation effect}: regime-dependent $\Rightarrow$
      UI dissipation falls with $\rho$ (via $\tilde A$), SS is
      ambiguous; only IU undistorted is \emph{insulated} at $\E[V_I]/2$
    \item \textbf{Threshold effect}: $\rho$ crosses $\rhosep$ $\Rightarrow$
      Riley distortion begins $\Rightarrow$ the low-type path is distorted,
      but welfare need not jump discontinuously at $\rhosep$
  \end{itemize}

  \begin{block}{Key implication}
    \begin{itemize}
      \item In IU undistorted separation, $\widetilde W^{\mathrm{sep}}$
        decreases with $\rho$ even though dissipation stays fixed
      \item Across regimes, welfare rankings remain parameter-dependent:
        SS and UI react through $\tilde A$, $\tilde B$,
        while Riley welfare must be judged from the full formula
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Equilibrium vs. Social Optimum}
  \begin{itemize}
    \setlength{\itemsep}{1em}
    \item \textbf{SS vs.\ UI}: welfare ranking is \emph{ambiguous} $\Rightarrow$
      sign of $(1-\tilde B)\bigl(\E[\sqrt{V_I}] - \tilde A/2\bigr)$;
      equal when $\tilde B = 1$
    \item \textbf{Undistorted sep.\ vs.\ Riley}: opposite responses to
      $\rho$ $\Rightarrow$ $\widetilde W^{\mathrm{sep}}$ falls while
      $\widetilde W^{\mathrm{Riley}}$ rises in $\rho$ $\Rightarrow$
      \emph{signal revelation and welfare improvement need not move
      together}
  \end{itemize}

  \vspace{0.4em}
  \begin{block}{Root cause}
    Equilibrium timing is pinned down purely by private payoff
    incentives; social welfare depends jointly on prize allocation
    \emph{and} effort dissipation $\Rightarrow$ different criteria.
    Equilibrium timing is therefore \emph{not} guaranteed to be socially
    optimal $\Rightarrow$ each regime's ranking must be checked from the full
    welfare formula.
  \end{block}
\end{frame}

\section{Conclusion}

\begin{frame}{Conclusion}
  \small
  \begin{enumerate}
    \setlength{\itemsep}{0.45em}
    \item \textbf{$\rho = 0$}: unique undistorted separation $\Rightarrow$
      $I$ leads at \emph{no revelation cost}.
    \item \textbf{$\rho > 0$} reopens the signal--belief--response
      channel: beliefs enter $U$'s best response.
    \item \textbf{Separation threshold}: undistorted iff
      $\rho \le \rhosep$; beyond, Riley distortion $\Rightarrow$
      cost borne by the low type alone.
    \item \textbf{Timing asymmetry}: $U$'s lead preference is robust;
      $I$ retreats from leading beyond $\rhodag$, opening a band where
      \emph{no} pure-strategy timing equilibrium survives.
    \item \citet{fu2006} is the $\rho = 1$ endpoint $\Rightarrow$
      not a general principle.
    \item \textbf{Welfare}: equilibrium timing can depart from the social
      optimum; responses to $\rho$ are regime-dependent.
    \item \textbf{Policy}: under correlated stakes (lobbying / rent-seeking),
      staged disclosure helps \emph{iff} IU welfare is higher $\Rightarrow$ else seal it.
  \end{enumerate}
\end{frame}
