\section{Introduction}

\begin{frame}{Research Question}
  \begin{itemize}
    \item \textbf{\citet{fu2006}}: in a Tullock contest with private information,
      the informed player prefers to move \emph{second} ---
      revealing type early is costly.
    \item But this rests on $\rho = 1$: prize values are
      \emph{perfectly correlated}.
    \item \textbf{Gap}: does the late-mover preference survive at $\rho < 1$,
      or is it an artifact of full correlation?
    \item \textbf{This paper}: introduce $\rho \in [0,1]$ ---
      at $\rho = 0$ the signal channel is closed;
      at $\rho = 1$ Fu's result is recovered as the limiting case.
  \end{itemize}

  \vfill
  \begin{center}
    \begin{tikzpicture}[scale=0.85, font=\footnotesize]
      \draw[->, thick] (0,0) -- (7.2,0) node[right] {$\rho$};

        \draw[thick] (\x,0.08) -- (\x,-0.08) node[below] {\lab};
      }
      \node[below=0.45cm, font=\scriptsize] at (6.8,0) {\citet{fu2006}};
    \end{tikzpicture}
  \end{center}
\end{frame}

\begin{frame}{What This Paper Does}
  \begin{itemize}
    \setlength{\itemsep}{1.2em}
    \item \textbf{\textit{i.i.d.} benchmark} ($\rho = 0$) ---
      isolates timing from information spillover.
    \item \textbf{Partial correlation} ($0 < \rho < 1$) as the main model
      --- $\rho = 1$ is the endpoint, not a separate model.
    \item Characterizes undistorted vs.\ Riley distorted separation,
      and their transition as $\rho$ rises.
    \item Links subgame equilibria to \textbf{endogenous timing}:
      a structural asymmetry between $I$ and $U$.
  \end{itemize}
\end{frame}

\section{Motivation and Benchmark}

\begin{frame}{Model Setup}
  \begin{itemize}
    \item Two players: informed ($I$) and uninformed ($U$)
    \item Types: $V_I, V_U \in \{v_H, v_L\}$,
      $\Pr(V_k = v_H) = q \in (0,1)$
    \item \textbf{Benchmark assumption}: $V_I \perp V_U$
      (\textit{i.i.d.}, $\rho = 0$)
  \end{itemize}

  \begin{center}
    \small
    \begin{tabular}{@{}lcc@{}}
      \toprule
      & $V_U = v_H$ & $V_U = v_L$ \\
      \midrule
      $V_I = v_H$ & $q^2$ & $q(1-q)$ \\
      $V_I = v_L$ & $q(1-q)$ & $(1-q)^2$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \begin{itemize}
    \item Three-stage timing: Stage 0 (choose timing) $\to$
      Nature draws types $\to$ effort subgame
    \item Three regimes: SS (simultaneous), UI ($U$ moves first),
      IU ($I$ moves first)
  \end{itemize}
\end{frame}

\begin{frame}{The Closed Signal Channel}
  \begin{block}{Lemma (Conditional Expectation Invariance)}
    If $V_I \perp V_U$, then
    \[
      \E[V_U \mid x_I] = \E[V_U],\; \forall x_I.
    \]
  \end{block}
  $\to$ \textbf{Signal channel closed}: beliefs update via Bayes,
  but $U$'s best response is unchanged ---
  \emph{updating happens, but it doesn't bite}.

  \vspace{0.6em}
  \begin{block}{Proposition (Unique Separating Equilibrium)}
    The IU subgame admits a \emph{unique separating equilibrium}, with IC slack
    \[
      \frac{(v_H - v_L)^2}{4\bar v} > 0,
      \qquad \bar v \equiv \E[V_k].
    \]
  \end{block}
  $\to$ $I$'s lead reveals her type at \emph{no cost}:
  the IC constraint never binds.
\end{frame}

\begin{frame}{Equilibrium Payoffs across Regimes}
  Composite coefficients:
  $\hat A \equiv \bar v\,\E\bigl[V_I^{-1/2}\bigr]$,
  $\hat B \equiv \bar v\,\E\bigl[V_I^{-1}\bigr] \ge 1$ (Jensen).

  \begin{center}
    \small
    \renewcommand{\arraystretch}{2.1}
    \begin{tabular}{@{}lccc@{}}
      \toprule
      Regime & $x_U^*$ & $\widetilde\pi_I$ & $\widetilde\pi_U$ \\
      \midrule
      SS
        & $\dfrac{\hat A^2}{(1+\hat B)^2}$
        & $\E\Bigl[\bigl(\sqrt{V_I} - \tfrac{\hat A}{1+\hat B}\bigr)^2\Bigr]$
        & $\dfrac{\hat A^2 \hat B}{(1+\hat B)^2}$ \\
      UI
        & $\dfrac{\hat A^2}{4}$
        & $\E\Bigl[\bigl(\sqrt{V_I} - \tfrac{\hat A}{2}\bigr)^2\Bigr]$
        & $\dfrac{\hat A^2}{4}$ \\
      IU (sep.)
        & $\dfrac{v(2\bar v - v)}{4\bar v}$
        & $\dfrac{\E[V_I^2]}{4\bar v}$
        & $\dfrac{\E\bigl[(2\bar v - V_I)^2\bigr]}{4\bar v}$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  $\to$ \textbf{First-mover advantage for $U$}:
  $\widetilde\pi_U^{UI} - \widetilde\pi_U^{SS}
    = \dfrac{\hat A^2 (1-\hat B)^2}{4(1+\hat B)^2} \ge 0$.
\end{frame}

\begin{frame}{Stage-0 Timing Game}
  Four unilateral deviation gaps:
  \[
    \Delta_I^{UI} \equiv \widetilde\pi_I^{UI} - \widetilde\pi_I^{SS},
    \quad
    \Delta_I^{IU} \equiv \widetilde\pi_I^{IU} - \widetilde\pi_I^{SS},
    \quad
    \Delta_U^{UI} \equiv \widetilde\pi_U^{UI} - \widetilde\pi_U^{SS},
    \quad
    \Delta_U^{SS} \equiv \widetilde\pi_U^{SS} - \widetilde\pi_U^{IU}
  \]

  \begin{block}{Proposition (Stage-0 Pure-strategy Equilibria)}
    Pure-strategy Nash equilibria are fully characterized
    by the signs of the four gaps:
    \begin{itemize}
      \item UI \; iff \; $\Delta_I^{UI} \ge 0$
      \item SS \; iff \; $\Delta_I^{UI} \le 0$ and $\Delta_U^{SS} \ge 0$
      \item IU \; iff \; $\Delta_I^{IU} \ge 0$ and $\Delta_U^{SS} \le 0$
    \end{itemize}
  \end{block}

  $\to$ \textbf{Takeaway}: timing is driven purely by gap signs ---
  no signaling distortion.

  $\to$ Any effect at $\rho > 0$ must come from the reopened channel ---
  not from asymmetric information per se.
\end{frame}

\section{Main Model}

\begin{frame}{Correlated Type Structure}
  \begin{itemize}
    \item Pearson correlation:
      $\rho \equiv \operatorname{corr}(H_I, H_U)$,
      where $H_k = \mathbf{1}\{V_k = v_H\}$, $\rho \in [0,1]$
    \item Marginals fixed at $(q, 1-q)$ --- $\rho$ is the single
      free parameter of the joint distribution
  \end{itemize}

  \begin{center}
    \small
    \begin{tabular}{@{}lcc@{}}
      \toprule
      & $V_U = v_H$ & $V_U = v_L$ \\
      \midrule
      $V_I = v_H$ & $q^2 + \rho q(1-q)$ & $q(1-q)(1-\rho)$ \\
      $V_I = v_L$ & $q(1-q)(1-\rho)$ & $(1-q)^2 + \rho q(1-q)$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \begin{center}
    \small
    \begin{tabular}{@{}lll@{}}
      \toprule
      $\rho$ & Structure & Key property \\
      \midrule
      $\rho = 0$ & \textit{i.i.d.} benchmark & signal has no spillover \\
      $0 < \rho < 1$ & partial correlation & belief enters $U$'s best response \\
      $\rho = 1$ & fully correlated & \citet{fu2006} endpoint \\
      \bottomrule
    \end{tabular}
  \end{center}
\end{frame}

\begin{frame}{Posterior Expected Valuation}
  Conditional probabilities of $U$'s high state, given $I$'s type:
  \[
    \theta_H = q + \rho(1-q),
    \qquad
    \theta_L = q(1-\rho).
  \]

  \begin{block}{Lemma (Posterior Expected Valuation)}
    Given posterior belief $\mu = \Pr(V_I = v_H \mid x_I)$,
    \[
      \VU(\mu) = v_L + \bigl[q + \rho(\mu - q)\bigr]\Delta V,
      \qquad
      \frac{d\VU(\mu)}{d\mu} = \rho\,\Delta V > 0.
    \]
  \end{block}

  \begin{center}
    \footnotesize
    \begin{tabular}{@{}llc@{}}
      \toprule
      Posterior $\mu$ & Interpretation & $\VU(\mu)$ \\
      \midrule
      $\mu = 1$ & $I$ is high type & $v_L + [q+\rho(1-q)]\Delta V$ \\
      $\mu = q$ & beliefs unchanged & $v_L + q\,\Delta V = \E[V_U]$ \\
      $\mu = 0$ & $I$ is low type & $v_L + q(1-\rho)\Delta V$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  $\to$ \textbf{When $\rho > 0$, $I$'s type is a signal about $U$'s own prize
  --- the channel is live.}
\end{frame}

\begin{frame}{SS and UI under Partial Correlation}
  Composite coefficients --- replace $\bar v$ with $\VU(1)$, $\VU(0)$:
  \[
    \tilde A \equiv q\,\frac{\VU(1)}{\sqrt{v_H}}
      + (1-q)\,\frac{\VU(0)}{\sqrt{v_L}},
    \qquad
    \tilde B \equiv q\,\frac{\VU(1)}{v_H}
      + (1-q)\,\frac{\VU(0)}{v_L}.
  \]

  \begin{itemize}
    \item Equilibrium structure identical in form to the benchmark:
      $\hat A, \hat B \to \tilde A, \tilde B$
      (at $\rho = 0$: $\tilde A = \hat A$, $\tilde B = \hat B$)
  \end{itemize}

  \begin{block}{Proposition (First-mover Advantage Robust)}
    $\widetilde\pi_U^{UI} \ge \widetilde\pi_U^{SS}$ for all $\rho \in [0,1)$.
  \end{block}

  \begin{itemize}
    \item \textbf{Monotone coefficients}: $\tilde A$, $\tilde B$ strictly
      decreasing in $\rho$, with $\tilde B \to 1$ as $\rho \to 1$ ---
      rising $\rho$ erases the Jensen excess burden $\hat B > 1$
  \end{itemize}
\end{frame}

\begin{frame}{When Does Separation Hold?}
  Undistorted separating candidate in the IU subgame, and its IC condition:
  \[
    x_H = \frac{v_H^2}{4\VU(1)},
    \quad
    x_L = \frac{v_L^2}{4\VU(0)};
    \qquad
    \frac{\VU(1)}{\VU(0)} \le R
    \equiv \frac{v_H^2}{v_L(2v_H - v_L)}.
  \]

  \begin{block}{Proposition (Separation Threshold)}
    Undistorted separation holds iff $\rho \le \rhosep$, where
    \[
      \rhosep = \frac{(R-1)\,\VU(q)}{(1-q+Rq)\,\Delta V}.
    \]
  \end{block}

  \begin{itemize}
    \item At $\rho = 0$: strictly satisfied; at $\rho = 1$: violated
    \item IC slack $S(\rho) \equiv R - \VU(1)/\VU(0)$ decreases
      monotonically and hits $0$ exactly at $\rhosep$
  \end{itemize}
\end{frame}

\begin{frame}{Riley Distortion}
  When $\rho > \rhosep$, define
  \[
    \alpha \equiv 1 - \frac{\VU(0)}{\VU(1)} \in (0,1),
    \qquad
    z_H^- \equiv \frac{v_H}{2\sqrt{\VU(0)}}\bigl(1 - \sqrt{\alpha}\bigr).
  \]

  \begin{itemize}
    \item High type keeps undistorted effort $x_H$;
      low type distorts \emph{down} to $x_L^{R} = (z_H^-)^2$
  \end{itemize}

  \begin{block}{Corollary (Two-region Structure)}
    \begin{itemize}
      \item $\rho \le \rhosep$: undistorted separation --- no signaling cost
      \item $\rho > \rhosep$: Riley distorted separation ---
        separation cost borne by the low type alone
    \end{itemize}
  \end{block}

  $\to$ High type's mimicry gain grows with $\rho$;
  the low type must undercut to deter it.
\end{frame}

\begin{frame}{Stage-0 Payoff Matrix}
  \textbf{Key point}: at stage 0, types are unrealized ---
  players compare \emph{ex ante} expected payoffs.

  \begin{center}
    \small
    \begin{tabular}{@{}lcc@{}}
      \toprule
      & $U$: Early & $U$: Late \\
      \midrule
      $I$: Early
        & $\bigl(\widetilde\pi_I^{SS},\, \widetilde\pi_U^{SS}\bigr)$
        & $\bigl(\widetilde\pi_I^{IU},\, \widetilde\pi_U^{IU}\bigr)$ \\
      $I$: Late
        & $\bigl(\widetilde\pi_I^{UI},\, \widetilde\pi_U^{UI}\bigr)$
        & $\bigl(\widetilde\pi_I^{SS},\, \widetilde\pi_U^{SS}\bigr)$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \begin{block}{Proposition (Stage-0 Pure-strategy Equilibria)}
    Same sign characterization as the benchmark
    ($\Delta_I^{UI}$, $\Delta_I^{IU}$, $\Delta_U^{UI}$, $\Delta_U^{SS}$)
    --- but now \emph{every entry depends on $\rho$}.
  \end{block}

  $\to$ The question becomes: \textbf{how do the gap signs move with $\rho$?}
\end{frame}

\begin{frame}{The Structural Asymmetry}
  \begin{itemize}
    \item \textbf{$U$'s side}: $\Delta_U^{UI} \ge 0$ for all $\rho < 1$
      --- $U$ always wants to lead against a late-moving $I$ (robust)
    \item \textbf{$I$'s side}: $\widetilde\pi_I^{IU}$ strictly decreasing
      in $\rho$ --- revelation cost rises with correlation
  \end{itemize}

  \begin{block}{Proposition (Timing Reversal)}
    If $\Delta_I^{IU}(0) > 0$ and $\Delta_I^{IU}(\rhosep) < 0$,
    there exists $\rhodag \in (0, \rhosep)$ where $I$'s timing
    preference flips from early to late.
  \end{block}

  \begin{center}
    \begin{tikzpicture}[scale=0.8, font=\footnotesize]
      \draw[->] (0,-1.1) -- (0,1.2) node[above] {$\Delta_I^{IU}(\rho)$};
      \draw[->] (0,0) -- (6.6,0) node[right] {$\rho$};
      \draw[thick] (0,0.8) .. controls (2.2,0.55) and (3.6,-0.25) .. (5.6,-0.85);
      \draw[thick] (3.1,0.07) -- (3.1,-0.07) node[below] {$\rhodag$};
      \draw[thick] (5.6,0.07) -- (5.6,-0.07) node[below] {$\rhosep$};
      \node[left, font=\scriptsize] at (0,0.8) {$>0$};
    \end{tikzpicture}
  \end{center}

  $\to$ \textbf{The reversal is driven entirely by $I$, not $U$.}
\end{frame}

\section{Welfare Analysis}

\begin{frame}{Welfare Framework}
  Social welfare = allocative value minus effort dissipation:
  $\widetilde W = \widetilde G - \widetilde T$,
  where $\widetilde G = \E[p_I V_I + p_U V_U]$
  and $\widetilde T = \E[x_I + x_U]$.

  \begin{center}
    \footnotesize
    \renewcommand{\arraystretch}{1.9}
    \begin{tabular}{@{}lcl@{}}
      \toprule
      Equilibrium type & Expected dissipation $\widetilde T$
        & Sensitivity to $\rho$ \\
      \midrule
      SS
        & $\dfrac{\tilde A}{1+\tilde B}\,\E\bigl[\sqrt{V_I}\bigr]$
        & varies via $\tilde A$, $\tilde B$ \\
      UI
        & $\dfrac{\tilde A}{2}\,\E\bigl[\sqrt{V_I}\bigr]$
        & varies via $\tilde A$ \\
      IU undistorted sep.
        & $\dfrac{\E[V_I]}{2}$
        & \textbf{immune to $\rho$} \\
      IU Riley distorted
        & $\dfrac{v_H}{2}\bigl[1-(1-q)\sqrt{\alpha}\bigr]$
        & varies via $\alpha$ \\
      IU pooling
        & $\sqrt{x_P\,\bar v}$
        & unaffected by $\rho$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  $\to$ \textbf{IU undistorted separation's dissipation is insulated
  from $\rho$} --- the key structural difference.
\end{frame}

\begin{frame}{Three Effects of $\rho$ on Welfare}
  \begin{itemize}
    \item \textbf{Allocative effect}: $\rho\uparrow$ $\to$
      $\VU(1)$ and $\VU(0)$ diverge $\to$ type-specific competition
      in IU matches true values better $\to$ allocation improves
    \item \textbf{Dissipation effect}: $\rho\uparrow$ $\to$
      SS dissipation keeps rising, while IU undistorted dissipation
      stays fixed at $\E[V_I]/2$ $\to$ IU's cost advantage widens
    \item \textbf{Threshold effect}: $\rho$ crosses $\rhosep$ $\to$
      Riley distortion begins $\to$ low-type allocation deteriorates
      --- a \emph{discontinuous} welfare drop
  \end{itemize}

  \vspace{0.4em}
  \begin{block}{$W(\rho)$ is generally non-monotone}
    \begin{itemize}
      \item Increasing for $\rho \le \min(\rhodag, \rhosep)$:
        dissipation is insulated, only the allocative effect operates
      \item Once the equilibrium shifts to SS at $\rhodag$,
        the increase stops
      \item Riley distortion beyond $\rhosep$ further reduces welfare
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Equilibrium vs. Social Optimum}
  \begin{itemize}
    \setlength{\itemsep}{1em}
    \item \textbf{Direction 1} ($\rho$ large): equilibrium is SS,
      yet $\widetilde W^{\mathrm{sep}} > \widetilde W^{SS}$ ---
      \emph{$I$'s retreat sustains the high-dissipation regime}
    \item \textbf{Direction 2} ($\rho$ small): equilibrium is IU,
      yet $\widetilde W^{SS} > \widetilde W^{IU}$ ---
      \emph{Riley distortion degrades the low-type allocation}
  \end{itemize}

  \vspace{0.4em}
  \begin{block}{Root cause}
    $I$ does not internalize $U$'s payoff; private timing incentives
    ignore the global dissipation--allocation trade-off.
    Efficiency of equilibrium timing is \emph{not} a general result ---
    it depends on the parameter configuration.
  \end{block}
\end{frame}

\section{Conclusion}

\begin{frame}{Conclusion}
  \small
  \begin{enumerate}
    \setlength{\itemsep}{0.9em}
    \item \textbf{$\rho = 0$}: unique undistorted separation ---
      $I$ leads at \emph{no revelation cost}.
    \item \textbf{$\rho > 0$} reopens the signal--belief--response
      channel: beliefs enter $U$'s best response.
    \item \textbf{Separation threshold}: undistorted iff
      $\rho \le \rhosep$; beyond, Riley distortion ---
      cost borne by the low type alone.
    \item \textbf{Timing asymmetry}: $U$'s lead preference is robust;
      $I$ flips to late at $\rhodag < \rhosep$.
    \item \textbf{\citet{fu2006}} is the $\rho = 1$ endpoint ---
      not a general principle.
    \item \textbf{Welfare}: equilibrium timing departs from the
      social optimum; $W(\rho)$ is non-monotone.
  \end{enumerate}
\end{frame}

\begin{frame}{Future Directions}
  \begin{enumerate}
    \setlength{\itemsep}{1.4em}
    \item \textbf{Continuous type space} --- beyond the closed-form
      threshold: richer distortion patterns.
    \item \textbf{Negative correlation} ($\rho < 0$) ---
      reverse information spillover.
    \item \textbf{$N$-player contests} --- timing symmetry-breaking;
      the role of correlation remains open.
  \end{enumerate}
\end{frame}
