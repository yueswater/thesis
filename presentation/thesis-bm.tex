\section{Introduction}

\begin{frame}{Research Question}
  \begin{itemize}
    \item How does prize-value correlation affect equilibrium revelation?
    \item How does it reshape the informed player's timing incentive?
    \item Can the benchmark and the fully correlated case be unified in one model?
  \end{itemize}

  \vspace{0.8em}
  \textbf{Main message}

  \begin{itemize}
    \item Correlation is the single force that moves the game from separation to pooling.
    \item The fully correlated case is the endpoint \(\rho = 1\), not a separate model.
  \end{itemize}
\end{frame}

\begin{frame}{What This Thesis Does}
  \begin{itemize}
    \item Replaces the old benchmark with an \textbf{iid benchmark}.
    \item Promotes \textbf{partial correlation} to the main model.
    \item Characterizes separation, pooling, and coexistence.
    \item Links subgame equilibrium to \textbf{endogenous timing}.
  \end{itemize}
\end{frame}

\begin{frame}{Correlation and the Main Mechanism}
  \small
  \begin{itemize}
    \item A single parameter \(\rho \in [0,1]\) indexes correlation:
    \[
      \Pr(V_I = v_H, V_U = v_H)=q^2+\rho q(1-q).
    \]
  \end{itemize}

  \begin{center}
    \begin{tabular}{>{\bfseries}m{2.2cm} m{7.8cm}}
      \toprule
      Case & Interpretation \\
      \midrule
      \(\rho=0\) & iid benchmark \\
      \(0<\rho<1\) & partial correlation \\
      \(\rho=1\) & fully correlated endpoint \\
      \bottomrule
    \end{tabular}
  \end{center}

  \vspace{0.6em}
  \begin{itemize}
    \item With posterior \(m(x_I)=\Pr(V_I=v_H\mid x_I)\), the effective prize is
    \[
      \hat V_U(m)=v_L+[q+\rho(m-q)](v_H-v_L),
    \]
    \item and the best response is
    \[
      x_U^*(x_I,m) = \sqrt{x_I \hat V_U(m)} - x_I.
    \]
  \end{itemize}
\end{frame}

\begin{frame}{Threshold Results}
  \small
  \begin{center}
    \begin{tabular}{>{\bfseries}m{2.8cm} m{3.1cm} m{4.3cm}}
      \toprule
      Result & Threshold & Meaning \\
      \midrule
      Undistorted separation & \(\rho \le \rhosep\) & survives only below a finite cutoff \\
      Pooling & \(\rho \ge \rhopool\) & appears before separation fully disappears \\
      Coexistence & \(\rho \in [\rhopool,\rhosep]\) & separation and pooling coexist \\
      \bottomrule
    \end{tabular}
  \end{center}

  \vspace{0.8em}
  \[
    \text{Benchmark calibration } (v_H,v_L,q)=(2,1,0.5):
    \qquad
    \rhopool=0.375,\quad \rhosep \approx 0.43.
  \]
\end{frame}

\begin{frame}{Conclusion}
  \begin{itemize}
    \item The iid benchmark and the fully correlated case belong to one unified model.
    \item Correlation is the key parameter behind:
    \begin{itemize}
      \item separation \(\rightarrow\) pooling,
      \item multiplicity and refinement,
      \item the informed player's timing preference.
    \end{itemize}
    \item The endpoint result is the limiting case of a broader equilibrium spectrum.
  \end{itemize}
\end{frame}
