## Key mathematical facts (pre-verified, do not re-derive)

Baseline: V_H=2, V_L=1, q=0.5

| Symbol | Formula | Value |
|--------|---------|-------|
| R | V_H²/[V_L(2V_H−V_L)] | 4/3 |
| ρ* | (R−1)V̂_U(q) / [ΔV(1−q+Rq)] | 3/7 ≈ 0.429 |
| ρ_P | β²V̂_U(q) / [(1−q)ΔV(1−β²)], β=(V_H−V_L)/(V_H+V_L) | 3/8 = 0.375 |
| V̂_U(0)\|ρ=ρ* | V_L + q(1−ρ*)ΔV | 9/7 |
| V̂_U(1)\|ρ=ρ* | V_L + [q+ρ*(1−q)]ΔV | 12/7 |
| α_sep\|ρ=ρ* | 1 − V̂_U(0)/V̂_U(1) | 1/4, √α_sep = 1/2 |
| x_L^Riley\|ρ=ρ* | V_H²/[4V̂_U(0)] · (1+√α_sep)² | 7/4 |

**Critical finding:** V_H=2=2V_L is the Case I/II boundary. Low-type deviation payoff = V_H(2V_L−V_H)/[4V̂_U(1)] = 0, so the low-type IC is marginally satisfied for *all* ρ. Distorted separating equilibrium therefore exists for all ρ ∈ (0,1], giving **ρ** = 1 and an empty Region II for baseline params.

Four-region structure (replaces old three-region):

| Region | ρ range | Equilibrium |
|--------|---------|-------------|
| I | [0, ρ_P) | Undistorted separating only |
| C | [ρ_P, ρ*] | Undistorted sep + pooling coexist |
| C' | (ρ*, 1] | Distorted sep (Riley) + pooling coexist |
| II | (ρ**, 1] | Pure pooling only — **empty for V_H=2V_L** |

```python
#!/usr/bin/env python3
"""Verify C1 (distorted separating) and C2 (correlated stage-0) math for v14."""
import numpy as np

def vhat(m, rho, V_H=2.0, V_L=1.0, q=0.5):
    dV = V_H - V_L
    return V_L + (q + rho*(m - q))*dV

def verify_baseline(V_H=2.0, V_L=1.0, q=0.5):
    dV = V_H - V_L
    R = V_H**2 / (V_L*(2*V_H - V_L))
    vq = vhat(q, 0, V_H, V_L, q)  # = V_L + q*dV, rho-independent
    rho_star = (R - 1)*vq / (dV*(1 - q + R*q))
    beta = dV / (V_H + V_L)
    rho_P = beta**2 * vq / ((1-q)*dV*(1 - beta**2))

    # At rho = rho_star
    v0 = vhat(0, rho_star, V_H, V_L, q)
    v1 = vhat(1, rho_star, V_H, V_L, q)
    alpha_sep = 1 - v0/v1
    xL_riley = V_H**2 / (4*v0) * (1 + np.sqrt(alpha_sep))**2

    # Low-type deviation payoff (cases I and II boundary check)
    low_dev = V_H*(2*V_L - V_H) / (4*v1)

    # Riley outcome: high-type IC at equality => x_L = xL_riley
    # Verify high-type IC equality
    pi_H_sep = V_H**2 / (4*v1)
    z_riley = np.sqrt(xL_riley)
    pi_H_dev_at_riley = V_H * z_riley / np.sqrt(v0) - xL_riley
    hi_ic_slack = pi_H_sep - pi_H_dev_at_riley

    print(f"=== V_H={V_H}, V_L={V_L}, q={q} ===")
    print(f"R          = {R:.6f}  (expected 4/3 ≈ 1.333333 for baseline)")
    print(f"rho_star   = {rho_star:.6f}  (expected 3/7 ≈ 0.428571)")
    print(f"rho_P      = {rho_P:.6f}  (expected 3/8 = 0.375000)")
    print(f"V_hat(0)|rho* = {v0:.6f}  (expected 9/7 ≈ 1.285714)")
    print(f"V_hat(1)|rho* = {v1:.6f}  (expected 12/7 ≈ 1.714286)")
    print(f"alpha_sep  = {alpha_sep:.6f}  (expected 1/4 = 0.250000)")
    print(f"sqrt(alpha)= {np.sqrt(alpha_sep):.6f}  (expected 1/2 = 0.500000)")
    print(f"x_L^Riley  = {xL_riley:.6f}  (expected 7/4 = 1.750000)")
    print(f"Low-type deviation payoff = {low_dev:.6f}  (expected 0 for V_H=2V_L)")
    print(f"High-type IC slack at Riley = {hi_ic_slack:.8f}  (expected ≈ 0)")
    print()

    # Check distorted sep survives all rho for this case (V_H = 2*V_L boundary)
    rho_vals = np.linspace(0.001, 0.999, 1000)
    always_exists = True
    for rho in rho_vals:
        v0r = vhat(0, rho, V_H, V_L, q)
        v1r = vhat(1, rho, V_H, V_L, q)
        alpha_r = 1 - v0r/v1r
        z_plus = V_H/(2*np.sqrt(v0r)) * (1 + np.sqrt(alpha_r))
        # Low-type IC: check deviation payoff vs min possible sep payoff
        ldev = V_H*(2*V_L - V_H)/(4*v1r)  # = 0 for V_H=2V_L always
        if ldev > 0:
            always_exists = False
            break
    print(f"Low-type IC auto-satisfied for all rho? {always_exists} (expected True for V_H=2V_L)")
    print(f"=> rho** = 1, Region II empty for baseline params")
    print()

def verify_general_case(V_H=3.0, V_L=1.0, q=0.5):
    """V_H > 2*V_L case: distorted sep exists for all rho (Case I)."""
    dV = V_H - V_L
    R = V_H**2 / (V_L*(2*V_H - V_L))
    vq = V_L + q*dV
    rho_star = (R - 1)*vq / (dV*(1 - q + R*q))
    beta = dV / (V_H + V_L)
    rho_P = beta**2 * vq / ((1-q)*dV*(1 - beta**2))
    low_dev_sign = V_H*(2*V_L - V_H)  # negative for V_H > 2*V_L

    print(f"=== V_H={V_H}, V_L={V_L}, q={q} (V_H > 2V_L case) ===")
    print(f"R          = {R:.6f}")
    print(f"rho_star   = {rho_star:.6f}")
    print(f"rho_P      = {rho_P:.6f}")
    print(f"V_H*(2V_L-V_H) = {low_dev_sign:.2f}  (negative => low-type IC auto-satisfied)")
    print(f"=> rho** = 1, distorted sep exists for all rho in (0,1]")
    print()

if __name__ == "__main__":
    verify_baseline()
    verify_general_case()
```

```python
def render_pooling_regions(
    V_H: float = 2.0,
    V_L: float = 1.0,
    q: float = 0.5,
) -> None:
    dV = V_H - V_L
    R = V_H**2 / (V_L * (2.0 * V_H - V_L))
    vhat_q = V_L + q * dV

    rho_star = (R - 1.0) * vhat_q / (dV * (1.0 - q + R * q))
    beta = (V_H - V_L) / (V_H + V_L)
    rho_P = beta**2 * vhat_q / ((1.0 - q) * dV * (1.0 - beta**2))
    # rho** = 1 for V_H = 2*V_L (baseline); distorted sep survives to rho=1
    rho_dstar = 1.0  # represents upper edge of C' = (rho*, rho**)

    fig, ax = plt.subplots(figsize=(7.2, 2.2))

    bar_y, bar_h = 0.0, 1.0
    segments = [
        (0.0,        rho_P,     "#F0F0F0", "I"),
        (rho_P,      rho_star,  "#D0D0D0", "C"),
        (rho_star,   rho_dstar, "#A0A0A0", "C'"),
        # Region II is empty for baseline (rho** = 1)
    ]
    for x0, x1, color, label in segments:
        if x1 <= x0:
            continue
        ax.add_patch(
            plt.Rectangle((x0, bar_y), x1 - x0, bar_h,
                          facecolor=color, edgecolor="black",
                          linewidth=0.7, zorder=1)
        )
        ax.text(0.5 * (x0 + x1), bar_y + 0.5 * bar_h, label,
                ha="center", va="center", fontsize=13, zorder=3,
                color=("white" if label == "II" else "black"))

    for xb in (rho_P, rho_star):
        ax.plot([xb, xb], [bar_y, bar_y + bar_h],
                color="black", linewidth=1.0, zorder=2)

    # rho_P: annotate BELOW the bar
    ax.annotate(
        fr"$\rho_P \approx {rho_P:.3f}$",
        xy=(rho_P, bar_y),
        xytext=(rho_P - 0.09, bar_y - 0.48),
        ha="center", va="top", fontsize=9,
        arrowprops=dict(arrowstyle="->", lw=0.7, color="black",
                        connectionstyle="arc3,rad=0.0"),
    )

    # rho*: annotate ABOVE the bar
    ax.annotate(
        fr"$\rho^* \approx {rho_star:.3f}$",
        xy=(rho_star, bar_y + bar_h),
        xytext=(rho_star + 0.09, bar_y + bar_h + 0.48),
        ha="center", va="bottom", fontsize=9,
        arrowprops=dict(arrowstyle="->", lw=0.7, color="black",
                        connectionstyle="arc3,rad=0.0"),
    )

    ax.text(0.0, bar_y + bar_h + 0.12, "0\n(independent)",
            ha="center", va="bottom", fontsize=8.5)
    ax.text(1.0, bar_y + bar_h + 0.12, r"1$(\rho^{**})$" + "\n(fully correlated)",
            ha="center", va="bottom", fontsize=8.5)

    ax.text(0.5 * rho_P, bar_y + bar_h + 0.12,

    ax.text(0.5 * (rho_P + rho_star), bar_y + bar_h + 0.12,

    ax.text(0.5 * (rho_star + 1.0), bar_y + bar_h + 0.12,
```
